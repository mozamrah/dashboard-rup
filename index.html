        function cleanNumber(val) {

            if (!val) return 0;

            if (typeof val === 'number') return val;

            let num = val.toString().replace(/[^0-9.-]+/g, "");

            return parseFloat(num) || 0;

        }


        function parseMonth(monthStr) {

            if(!monthStr) return 0;

            if(!isNaN(monthStr)) return parseInt(monthStr); 

            const m = monthStr.toString().toLowerCase();

            if(m.includes('jan')) return 1; if(m.includes('feb')) return 2; if(m.includes('mac') || m.includes('mar')) return 3;

            if(m.includes('apr')) return 4; if(m.includes('mei') || m.includes('may')) return 5; if(m.includes('jun')) return 6;

            if(m.includes('jul')) return 7; if(m.includes('ogo') || m.includes('aug')) return 8; if(m.includes('sep')) return 9;

            if(m.includes('okt') || m.includes('oct')) return 10; if(m.includes('nov')) return 11; if(m.includes('dis') || m.includes('dec')) return 12;

            return 0;

        }

        function extractDataFromGrid(grid) {

            let headerRowIndex = -1;

            let colMap = { 

                agensi: -1, bulan: -1, kategori: -1, 

                rKursus: -1, sKursus: -1, kursus: -1, 

                rPeserta: -1, sPeserta: -1, peserta: -1, 

                rKos: -1, sKos: -1, kos: -1 

            };


            const kwAgensi = ['agensi', 'jabatan', 'institusi', 'institut', 'iljtm', 'pusat', 'nama'];

            const kwBulan = ['bulan', 'bln', 'tarikh', 'month', 'tempoh'];

            const kwKategori = ['kategori', 'jenis', 'status', 'perancangan', 'pelaksanaan', 'tahap'];


            for (let i = 0; i < Math.min(grid.length, 15); i++) {

                if (!grid[i]) continue;

                let rowJoined = grid[i].join('').toLowerCase().replace(/[^a-z0-9]/g, '');

                

                if (kwAgensi.some(k => rowJoined.includes(k)) || kwBulan.some(k => rowJoined.includes(k))) {

                    headerRowIndex = i;

                    grid[i].forEach((col, index) => {

                        let cleanCol = col.toLowerCase().replace(/[^a-z0-9]/g, '');

                        if (kwAgensi.some(k => cleanCol.includes(k))) colMap.agensi = index;

                        else if (kwBulan.some(k => cleanCol.includes(k))) colMap.bulan = index;

                        else if (kwKategori.some(k => cleanCol.includes(k))) colMap.kategori = index;

                        

                        else if (cleanCol.includes('kursus') || cleanCol.includes('siri') || cleanCol.includes('program')) {

                            if (cleanCol.includes('rancang') || cleanCol.includes('plan')) colMap.rKursus = index;

                            else if (cleanCol.includes('sebenar') || cleanCol.includes('laksana')) colMap.sKursus = index;

                            else colMap.kursus = index;

                        }

                        else if (cleanCol.includes('peserta') || cleanCol.includes('pengajar') || cleanCol.includes('orang')) {

                            if (cleanCol.includes('rancang') || cleanCol.includes('plan')) colMap.rPeserta = index;

                            else if (cleanCol.includes('sebenar') || cleanCol.includes('laksana')) colMap.sPeserta = index;

                            else colMap.peserta = index;

                        }

                        else if (cleanCol.includes('kos') || cleanCol.includes('rm') || cleanCol.includes('belanja') || cleanCol.includes('peruntukan')) {

                            if (cleanCol.includes('rancang') || cleanCol.includes('plan') || cleanCol.includes('peruntukan')) colMap.rKos = index;

                            else if (cleanCol.includes('sebenar') || cleanCol.includes('laksana') || cleanCol.includes('belanja')) colMap.sKos = index;

                            else colMap.kos = index;

                        }

                    });

                    break;

                }

            }


            let results = [];

            if (headerRowIndex === -1 || colMap.agensi === -1 || colMap.bulan === -1) {

                return results;

            }


            for (let i = headerRowIndex + 1; i < grid.length; i++) {

                let row = grid[i];

                let agensi = row[colMap.agensi];

                let bulanRaw = row[colMap.bulan];


                if (!agensi || !bulanRaw) continue; 

                if (agensi.toLowerCase().includes('jumlah') || agensi.toLowerCase().includes('total')) continue; 


                let bulanNum = parseMonth(bulanRaw);

                if (bulanNum === 0) continue;


                let isRancang = true;

                let isSebenar = true;


                if (colMap.kategori > -1) {

                    let kat = String(row[colMap.kategori]).toLowerCase();

                    if (kat.includes('rancang') || kat.includes('plan')) {

                        isRancang = true; isSebenar = false;

                    } else if (kat.includes('sebenar') || kat.includes('laksana')) {

                        isRancang = false; isSebenar = true;

                    }

                }


                results.push({

                    agensi: agensi.trim().toUpperCase(),

                    bulan: bulanNum,

                    rKursus: colMap.rKursus > -1 ? cleanNumber(row[colMap.rKursus]) : (isRancang && colMap.kursus > -1 ? cleanNumber(row[colMap.kursus]) : 0),

                    sKursus: colMap.sKursus > -1 ? cleanNumber(row[colMap.sKursus]) : (isSebenar && colMap.kursus > -1 ? cleanNumber(row[colMap.kursus]) : 0),

                    rPeserta: colMap.rPeserta > -1 ? cleanNumber(row[colMap.rPeserta]) : (isRancang && colMap.peserta > -1 ? cleanNumber(row[colMap.peserta]) : 0),

                    sPeserta: colMap.sPeserta > -1 ? cleanNumber(row[colMap.sPeserta]) : (isSebenar && colMap.peserta > -1 ? cleanNumber(row[colMap.peserta]) : 0),

                    rKos: colMap.rKos > -1 ? cleanNumber(row[colMap.rKos]) : (isRancang && colMap.kos > -1 ? cleanNumber(row[colMap.kos]) : 0),

                    sKos: colMap.sKos > -1 ? cleanNumber(row[colMap.sKos]) : (isSebenar && colMap.kos > -1 ? cleanNumber(row[colMap.kos]) : 0)

                });

            }

            return results;

        }


        async function loadData() {

            try {

                document.getElementById('sync-status-indicator').className = "w-2 h-2 rounded-full bg-yellow-400 animate-pulse";

                document.getElementById('sync-status-text').innerText = "Menarik data Google Sheets...";

                document.getElementById('debug-section').classList.add('hidden');

                document.getElementById('debug-content').innerHTML = '';


                const dataArray = await fetchCSV(URL_DATA);


                let mergedMap = {};

                let parsedData = extractDataFromGrid(dataArray);


                if (parsedData.length === 0) {

                    document.getElementById('sync-status-indicator').className = "w-2 h-2 rounded-full bg-red-500";

                    document.getElementById('sync-status-text').innerText = "Format Jadual Tidak Dikenali";

                    renderDebugTable(dataArray, "Data Raw (Google Sheets)");

                    processData();

                    return;

                }


                parsedData.forEach(d => {

                    let key = `${d.agensi}_${d.bulan}`;

                    if(!mergedMap[key]) mergedMap[key] = { agensi: d.agensi, bulan: d.bulan, rKursus: 0, sKursus: 0, rPeserta: 0, sPeserta: 0, rKos: 0, sKos: 0 };

                    mergedMap[key].rKursus += d.rKursus;

                    mergedMap[key].sKursus += d.sKursus;

                    mergedMap[key].rPeserta += d.rPeserta;

                    mergedMap[key].sPeserta += d.sPeserta;

                    mergedMap[key].rKos += d.rKos;

                    mergedMap[key].sKos += d.sKos;

                });


                rawData = Object.values(mergedMap);


                document.getElementById('sync-status-indicator').className = "w-2 h-2 rounded-full bg-green-400";

                document.getElementById('sync-status-text').innerText = "Live: Google Sheets Sync";


                const filter = document.getElementById('agencyFilter');

                const rawAgencies = [...new Set(rawData.map(item => item.agensi))];

                const agencies = sortAgencies(rawAgencies); // Gunakan fungsi susunan khas

                let html = `<option value="ALL">Semua Agensi</option>`;

                agencies.forEach(ag => html += `<option value="${ag}">${ag}</option>`);

                

                const currentVal = filter.value;

                filter.innerHTML = html;

                if(agencies.includes(currentVal)) filter.value = currentVal;


                processData();


            } catch (error) {

                console.error("Ralat memuat turun:", error);

                document.getElementById('sync-status-indicator').className = "w-2 h-2 rounded-full bg-red-500";

                document.getElementById('sync-status-text').innerText = "Ralat Sambungan!";

            }

        }


        function refreshData() {

            const btn = document.querySelector('button');

            const originalContent = btn.innerHTML;

            btn.innerHTML = '<i class="fas fa-spinner fa-spin"></i> Menyemak...';

            loadData().then(() => { btn.innerHTML = originalContent; });

        }


        function processData() {

            const filterAgency = document.getElementById('agencyFilter').value;

            const startMonth = parseInt(document.getElementById('monthStart').value);

            const endMonth = parseInt(document.getElementById('monthEnd').value);


            // Logik pintar: Pastikan min dan max betul walau pengguna terbalik pilih bulan

            const minM = Math.min(startMonth, endMonth);

            const maxM = Math.max(startMonth, endMonth);


            // Saringan (Filter) berdasarkan julat bulan

            let monthFilteredData = rawData.filter(d => d.bulan >= minM && d.bulan <= maxM);


            // Saringan (Filter) berdasarkan agensi

            let fullyFilteredData = monthFilteredData;

            if (filterAgency !== 'ALL') {

                fullyFilteredData = monthFilteredData.filter(d => d.agensi === filterAgency);

            }


            let t_rKursus = 0, t_sKursus = 0, t_rPeserta = 0, t_sPeserta = 0, t_rKos = 0, t_sKos = 0;


            fullyFilteredData.forEach(d => {

                t_rKursus += d.rKursus; t_sKursus += d.sKursus;

                t_rPeserta += d.rPeserta; t_sPeserta += d.sPeserta;

                t_rKos += d.rKos; t_sKos += d.sKos;

            });


            document.getElementById('kpi-kursus-sebenar').innerText = formatNumber(t_sKursus);

            document.getElementById('kpi-kursus-rancang').innerText = formatNumber(t_rKursus);

            const pctKursus = t_rKursus > 0 ? Math.round((t_sKursus / t_rKursus) * 100) : 0;

            document.getElementById('kpi-kursus-pct').innerText = `${pctKursus}%`;

            document.getElementById('bar-kursus').style.width = `${Math.min(pctKursus, 100)}%`;


            document.getElementById('kpi-peserta-sebenar').innerText = formatNumber(t_sPeserta);

            document.getElementById('kpi-peserta-rancang').innerText = formatNumber(t_rPeserta);

            const pctPeserta = t_rPeserta > 0 ? Math.round((t_sPeserta / t_rPeserta) * 100) : 0;

            document.getElementById('kpi-peserta-pct').innerText = `${pctPeserta}%`;

            document.getElementById('bar-peserta').style.width = `${Math.min(pctPeserta, 100)}%`;


            document.getElementById('kpi-kos-sebenar').innerText = formatRM(t_sKos);

            document.getElementById('kpi-kos-rancang').innerText = formatRM(t_rKos);

            const pctKos = t_rKos > 0 ? Math.round((t_sKos / t_rKos) * 100) : 0;

            document.getElementById('kpi-kos-pct').innerText = `${pctKos}%`;

            document.getElementById('bar-kos').style.width = `${Math.min(pctKos, 100)}%`;


            updateMonthlyChart(fullyFilteredData);

            updateAgencyChart(monthFilteredData); 

            updateGanttChart(monthFilteredData, filterAgency);

        }


        function updateMonthlyChart(data) {

            const months = ['Jan', 'Feb', 'Mac', 'Apr', 'Mei', 'Jun', 'Jul', 'Ogo', 'Sep', 'Okt', 'Nov', 'Dis'];

            let rData = new Array(12).fill(0);

            let sData = new Array(12).fill(0);


            data.forEach(d => {

                if(d.bulan >= 1 && d.bulan <= 12) {

                    rData[d.bulan - 1] += d.rPeserta;

                    sData[d.bulan - 1] += d.sPeserta;

                }

            });


            const ctx = document.getElementById('monthlyChart').getContext('2d');

            if (monthlyChartInstance) monthlyChartInstance.destroy();


            monthlyChartInstance = new Chart(ctx, {

                type: 'bar',

                data: {

                    labels: months,

                    datasets: [

                        { label: 'Perancangan', data: rData, backgroundColor: 'rgba(156, 163, 175, 0.5)', borderRadius: 4 },

                        { label: 'Pelaksanaan Sebenar', data: sData, backgroundColor: 'rgba(59, 130, 246, 0.8)', borderRadius: 4 }

                    ]

                },

                options: { responsive: true, maintainAspectRatio: false, plugins: { legend: { position: 'top' } } }

            });

        }


        function updateAgencyChart(allData) {

            // Gunakan rawData untuk mendapatkan senarai penuh agensi supaya carta tidak mengecil

            const rawAgencies = [...new Set(rawData.map(d => d.agensi))];

            const agencies = sortAgencies(rawAgencies); // Gunakan fungsi susunan khas

            

            const metric = document.getElementById('agencyChartMetric').value;

            const isKos = metric === 'kos';

            

            let performance = [];

            let tooltipData = [];


            agencies.forEach(ag => {

                let plan = 0, actual = 0;

                allData.filter(d => d.agensi === ag).forEach(d => { 

                    if (metric === 'kursus') { plan += d.rKursus; actual += d.sKursus; }

                    else if (metric === 'peserta') { plan += d.rPeserta; actual += d.sPeserta; }

                    else if (metric === 'kos') { plan += d.rKos; actual += d.sKos; }

                });

                

                let pct = plan > 0 ? Math.round((actual/plan)*100) : 0;

                performance.push(pct);

                

                tooltipData.push({

                    pct: pct,

                    plan: plan,

                    actual: actual,

                    isKos: isKos

                });

            });


            const colors = ['#10B981', '#F59E0B', '#6366F1', '#EC4899', '#0EA5E9'];

            const ctx = document.getElementById('agencyChart').getContext('2d');

            if (agencyChartInstance) agencyChartInstance.destroy();


            agencyChartInstance = new Chart(ctx, {

                type: 'bar',

                data: { labels: agencies, datasets: [{ data: performance, backgroundColor: colors, borderRadius: 4 }] },

                options: { 

                    indexAxis: 'y', 

                    responsive: true, 

                    maintainAspectRatio: false, 

                    plugins: { 

                        legend: { display: false },

                        tooltip: {

                            callbacks: {

                                label: function(context) {

                                    const td = tooltipData[context.dataIndex];

                                    const valS = td.isKos ? formatRM(td.actual) : formatNumber(td.actual);

                                    const valP = td.isKos ? formatRM(td.plan) : formatNumber(td.plan);

                                    return `Pencapaian: ${td.pct}% (Sebenar: ${valS} / Rancang: ${valP})`;

                                }

                            }

                        }

                    }, 

                    scales: { 

                        x: { 

                            suggestedMax: 100, // Menggunakan suggestedMax supaya graf boleh pergi >100% jika terlebih laksana

                            title: { display: true, text: 'Peratusan Siap (%)' }

                        } 

                    } 

                }

            });

        }


        // Fungsi singkatan nombor untuk carta (Cth: 1500 -> 1.5K, 2000000 -> 2M)

        function formatShortNum(num) {

            if (!num || num === 0) return '0';

            if (num >= 1000000) return (num / 1000000).toFixed(1) + 'M';

            if (num >= 1000) return (num / 1000).toFixed(1) + 'K';

            return num.toString();

        }


        function updateGanttChart(allData, filterAgency) {

            const rawAgencies = [...new Set(rawData.map(d => d.agensi))];

            const allAgencies = sortAgencies(rawAgencies); 

            const agencies = filterAgency === 'ALL' ? allAgencies : [filterAgency];

            const ganttBody = document.getElementById('gantt-body');

            const metric = document.getElementById('ganttMetric').value;

            const isKos = metric === 'kos';

            

            ganttBody.innerHTML = ''; 


            if(agencies.length === 0) {

                ganttBody.innerHTML = '<p class="text-sm text-gray-500 py-4">Tiada data dijumpai.</p>';

                return;

            }


            agencies.forEach(ag => {

                const agData = allData.filter(d => d.agensi === ag);

                let htmlRow = `<div class="flex items-center text-sm py-2 hover:bg-gray-50 rounded transition border-b border-gray-100 last:border-0"><div class="w-32 flex-shrink-0 font-medium text-gray-700 truncate">${ag}</div><div class="flex-1 grid grid-cols-12 gap-1 h-11">`;


                for(let i=1; i<=12; i++) {

                    const mData = agData.find(d => d.bulan === i);

                    let rVal = 0, sVal = 0;

                    

                    if(mData) {

                        if(metric === 'kursus') { rVal = mData.rKursus; sVal = mData.sKursus; }

                        else if(metric === 'peserta') { rVal = mData.rPeserta; sVal = mData.sPeserta; }

                        else if(metric === 'kos') { rVal = mData.rKos; sVal = mData.sKos; }

                    }


                    if(rVal > 0 || sVal > 0) {

                        let pText = formatShortNum(rVal);

                        let sText = formatShortNum(sVal);

                        

                        let bgClassP = "bg-gray-200 text-gray-600"; // Warna Rancang (Default)

                        let bgClassS = "bg-gray-100 text-gray-400"; // Warna Sebenar (Default Kosong)


                        if (rVal > 0 && sVal === 0) {

                            bgClassS = "bg-red-400 text-white"; // Merah jika belum laksana tapi ada plan

                        } else if (rVal > 0 && sVal > 0) {

                            let ratio = sVal / rVal;

                            bgClassS = ratio >= 1 ? "bg-green-500 text-white" : ratio >= 0.5 ? "bg-yellow-400 text-gray-800" : "bg-orange-400 text-white";

                        } else if (rVal === 0 && sVal > 0) {

                            bgClassS = "bg-blue-400 text-white"; // Biru jika ad-hoc (tiada plan tapi laksana)

                            bgClassP = "bg-gray-100 text-gray-400";

                        }


                        let tooltipStr = `Rancang: ${isKos ? formatRM(rVal) : formatNumber(rVal)} \nSebenar: ${isKos ? formatRM(sVal) : formatNumber(sVal)}`;


                        htmlRow += `

                        <div class="flex flex-col h-full rounded shadow-sm overflow-hidden" title="${tooltipStr}">

                            <div class="${bgClassP} h-1/2 flex items-center justify-center text-[10px] font-medium border-b border-white/40">P: ${pText}</div>

                            <div class="${bgClassS} h-1/2 flex items-center justify-center text-[10px] font-bold tracking-tight">S: ${sText}</div>

                        </div>`;

                    } else {

                        htmlRow += `<div class="bg-transparent h-full"></div>`;

                    }

                }

                htmlRow += `</div></div>`;

                ganttBody.innerHTML += htmlRow;

            });

        }
